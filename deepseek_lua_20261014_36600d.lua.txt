local coordenadas={}
local function SalvarCoord(t)
    local s="" for n,c in pairs(t) do s=s..string.format("%s|%.2f,%.2f,%.2f\n",n,c.x,c.y,c.z) end
    pcall(function() writefile("ball_pro_coords.txt",s) end)
end
local function CarregarCoord()
    local ok,ct=pcall(function() return readfile("ball_pro_coords.txt") end)
    if not ok or not ct then return {} end
    local t={} for l in ct:gmatch("[^\n]+") do
        local n,xy=l:match("^(.+)|(.+)$")
        if n and xy then local x,y,z=xy:match("([^,]+),([^,]+),([^,]+)")
            if x and y and z then t[n]={x=tonumber(x),y=tonumber(y),z=tonumber(z)} end end
    end return t
end
coordenadas=CarregarCoord()

local function NomeParAlternativo(nome)
    if not nome or nome=="" then return nil end
    local lo=nome:lower()
    local pares={{"esquerdo","direito"},{"left","right"},{"esq","dir"},{"e","d"},{"l","r"},{"norte","sul"},{"cima","baixo"},{"top","bottom"},{"up","down"}}
    for _,p in ipairs(pares) do
        if lo:find(p[1],1,true) then return nome:gsub("(?i)"..p[1],p[2]) end
        if lo:find(p[2],1,true) then return nome:gsub("(?i)"..p[2],p[1]) end
    end return nil
end
local function TrocarAnguloCoord()
    if not estadoSistema.coordenadaSelecionada then Notify("Nenhuma coordenada","warn") return end

        estadoSistema.coordenadaSelecionada=a
        local c=coordenadas[a]
        estadoSistema.alvoAtual=Vector3.new(c.x,c.y,c.z)
        Notify("Ângulo → "..a,"ok")
    else Notify("Par não encontrado","warn") end
end

local function getBallPart(o)
    if not o then return nil end
    if o:IsA("BasePart") or o:IsA("UnionOperation") then return o end
    if o:IsA("Model") then return o.PrimaryPart or o:FindFirstChild("MainPart") or o:FindFirstChild("Ball") or o:FindFirstChild("TPS") or o:FindFirstChildWhichIsA("BasePart",true) end
    return nil
end
local function IsRealBall(o)
    if not o or not o.Parent or not o:IsA("BasePart") then return false end
    if Char and o:IsDescendantOf(Char) then return false end
    local n=o.Name:lower()
    if n=="head" or n=="torso" then return false end
    if n:find("stadium",1,true) or n:find("map",1,true) or n:find("goal",1,true) or n:find("net",1,true) or n:find("grass",1,true) or n:find("field",1,true) or n:find("pitch",1,true) then return false end
    local m=o:FindFirstAncestorOfClass("Model")
    if m and m:FindFirstChildOfClass("Humanoid") then return false end
    local s=o.Size
    if s.X<0.35 or s.X>10 or s.Y<0.35 or s.Y>10 or s.Z<0.35 or s.Z>10 then return false end
    local mn=m and m.Name:lower() or "" local pn=o.Parent and o.Parent.Name:lower() or ""
    if regNames[o.Name] or regNames[n] or regNames[m and m.Name or ""] or regNames[mn] or regNames[o.Parent and o.Parent.Name or ""] or regNames[pn] then return true end
    local function lk(x) return x=="tps" or x=="ball" or x=="bola" or x=="soccer ball" or x=="soccer_ball" or x:find("soccerball",1,true) or x:find("football",1,true) or (x:find("soccer",1,true) and x:find("ball",1,true)) or x:find("ballpart",1,true)~=nil end
    if lk(n) or lk(mn) or lk(pn) then return true end
    local mesh=o:FindFirstChildOfClass("SpecialMesh")
    if mesh and not o.Anchored then local mx=math.max(s.X,s.Y,s.Z) local mi=math.min(s.X,s.Y,s.Z) if mx/math.max(mi,0.01)<=2.5 then return true end end
    if o:IsA("MeshPart") and not o.Anchored then local mx=math.max(s.X,s.Y,s.Z) local mi=math.min(s.X,s.Y,s.Z) if mx/math.max(mi,0.01)<=2.5 then return true end end
    return false
end

local BallCache={} ActiveBall=nil ActiveBallLastSeen=0 BallReacquireGrace=1.75 BallScanInterval=0.08 BallLastScan=0
local function RefreshBallCache(force)
    local now=os.clock() if not force and now-BallLastScan<BallScanInterval then return end BallLastScan=now
    local seen={}
    for _,o in ipairs(workspace:GetDescendants()) do
        if IsRealBall(o) then
            seen[o]=true BallCache[o]=BallCache[o] or {lastSeen=now,lastPos=o.Position,lastSpeed=0}
            local d=BallCache[o] d.lastSeen=now d.lastPos=o.Position d.lastSpeed=o.AssemblyLinearVelocity.Magnitude
        end
    end
    for o,d in pairs(BallCache) do if not seen[o] and (not o.Parent or now-(d.lastSeen or 0)>2.5) then BallCache[o]=nil end end
end
local function GetAllBalls(incSlow)
    RefreshBallCache(false) local list={} local now=os.clock()
    for b,d in pairs(BallCache) do
        if b and b.Parent and IsRealBall(b) then
            local sp=b.AssemblyLinearVelocity.Magnitude
            local di=Root and (b.Position-Root.Position).Magnitude or math.huge
            if incSlow or sp>=5 or b==ActiveBall or now-(d.lastSeen or 0)<BallReacquireGrace then
                table.insert(list,{ball=b,speed=sp,dist=di,namePriority=(b.Name:lower()=="tps" or b.Name:lower()=="ball" or b.Name:lower()=="bola") and 3 or 1})
            end
        end
    end
    table.sort(list,function(a,b)
        if a.ball==ActiveBall then return true end if b.ball==ActiveBall then return false end
        return (a.namePriority*1000+a.speed*5-math.min(a.dist,300))>(b.namePriority*1000+b.speed*5-math.min(b.dist,300))
    end)
    return list
end

local cdToque=0
local function CalcularDesvio(b,dir)
    if not desvioOn then return dir end
    local mp,md=nil,math.huge local bp=b.Position
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LP and p.Character then
            local h=p.Character:FindFirstChild("HumanoidRootPart")
            if h then local d=(h.Position-bp).Magnitude
                if d<=raioDesvio and d<md then
                    local pp=Vector3.new(h.Position.X-bp.X,0,h.Position.Z-bp.Z)
                    if pp.Magnitude>0.1 and dir:Dot(pp.Unit)>0.3 then mp=h md=d end
                end end
        end
    end
    if not mp then return dir end
    local pp=Vector3.new(mp.Position.X-bp.X,0,mp.Position.Z-bp.Z).Unit
    local perp=Vector3.new(-dir.Z,0,dir.X)
    local ld=perp:Dot(pp)>0 and -perp or perp
    local u=math.clamp(1-(md/raioDesvio),0,1)
    local dd=dir:Lerp(ld,forcaDesvio*u)
    return dd.Magnitude>0.01 and dd.Unit or dir
end

end
local function GuideBallToPoint(b,tp,lt,st,mn,mx)
    if not b or not b.Parent or not tp then return end
    local v=b.AssemblyLinearVelocity local vh=Vector3.new(v.X,0,v.Z) local sp=vh.Magnitude
    if sp<(mn or 5) then return end
    local pr=b.Position+vh*(lt or 0.12)
    local ft=Vector3.new(tp.X,pr.Y,tp.Z)
    local tt=Vector3.new(ft.X-pr.X,0,ft.Z-pr.Z)
    if tt.Magnitude<0.1 then return end
    local cd=vh.Unit local td=tt.Unit
    local dt=math.clamp(cd:Dot(td),-1,1)
    local cor=math.clamp((st or 0.35)*(1-dt*0.30),0.035,0.48)
    local fd=cd:Lerp(td,cor) if fd.Magnitude<0.01 then fd=td end fd=fd.Unit
    local ds=math.clamp(sp+(tt.Magnitude>35 and 0.9 or 0),mn or 5,mx or 135)
    b.AssemblyLinearVelocity=Vector3.new(fd.X*ds,v.Y,fd.Z*ds)
    b.AssemblyAngularVelocity=Vector3.zero
end
local function GuiarBola(b,pg,dt)
    if not estadoSistema.ativo or not estadoSistema.alvoAtual then return end
    local v=b.AssemblyLinearVelocity local mg=Vector3.new(v.X,0,v.Z).Magnitude
    if not estadoSistema.guiaAtivo and not estadoSistema.cruzou and mg>=VEL_CHUTE_MIN then
        estadoSistema.guiaAtivo=true estadoSistema.contandoParada=0
        Notify(string.format("Chute! vel: %.0f",mg),"info")
    end
    if not estadoSistema.guiaAtivo then return end
    local dg=Vector3.new(pg.X-b.Position.X,0,pg.Z-b.Position.Z) local di=dg.Magnitude
    if estadoSistema.cruzou or di<=raioGoal then
        estadoSistema.cruzou=true
        local nv=b.AssemblyLinearVelocity*freioPosGol
        if nv.Magnitude<3 then
            b.AssemblyLinearVelocity=Vector3.zero b.AssemblyAngularVelocity=Vector3.zero
            estadoSistema.contandoParada=estadoSistema.contandoParada+1
            if estadoSistema.contandoParada>TEMPO_PARA_DESLIGAR then
                estadoSistema.guiaAtivo=false estadoSistema.cruzou=false estadoSistema.contandoParada=0 estadoSistema.ativo=false
                Notify("Alvo alcançado","ok")
            end return
        end b.AssemblyLinearVelocity=nv return
    end
    local vh=Vector3.new(v.X,0,v.Z)
    local ap=vh.Magnitude>1 and vh.Unit:Dot(dg.Unit)>0.4
    if di<=zonaGoal and ap then
        local nm=math.max(vh.Magnitude-ATRITO_GOAL*dt,0)
        if nm*dt*60<di then
            b.AssemblyLinearVelocity=Vector3.zero b.AssemblyAngularVelocity=Vector3.zero
            estadoSistema.contandoParada=estadoSistema.contandoParada+1
            if estadoSistema.contandoParada>TEMPO_PARA_DESLIGAR then
                estadoSistema.guiaAtivo=false estadoSistema.cruzou=false estadoSistema.contandoParada=0 estadoSistema.ativo=false
            end return
        end
        b.AssemblyLinearVelocity=Vector3.new(vh.Unit.X*nm,v.Y,vh.Unit.Z*nm) return
    end
    local es=FORCA_CURVA
    if di<DIST_INICIO_CURVA then es=es*(0.78+math.clamp(1-di/DIST_INICIO_CURVA,0,1)*0.30) end
    if di<22 then es=es*0.85 end
    GuideBallToPoint(b,pg,0.12,es,VEL_MINIMA,135)
    if os.clock()-cdToque>0.15 then cdToque=os.clock() TocarBola(b) end
end

local function DoSkill(dt)
    if not SkillModeEnabled or not Root or not Root.Parent then return end
    local cl=GetValidBallFollow() if not cl or not cl.Parent then return end
    if (cl.Position-Root.Position).Magnitude>12 then return end
    skillTimer=skillTimer+(dt or 0.016)*SkillSpeed
    if skillTimer>=1 then skillTimer=0 skillPhase=(skillPhase+1)%4 end
    local t=skillTimer local off
    if skillPhase==0 then off=Root.CFrame.LookVector*math.sin(t*math.pi)*2
    elseif skillPhase==1 then off=Root.CFrame.RightVector*math.sin(t*math.pi)*2
    elseif skillPhase==2 then off=-Root.CFrame.RightVector*math.sin(t*math.pi)*2
    else off=-Root.CFrame.LookVector*math.sin(t*math.pi)*2 end
    local tp=Root.Position+off+Vector3.new(0,-1.5,0)
    cl.CFrame=CFrame.new(cl.Position:Lerp(tp,0.3),cl.Position)
    cl.AssemblyLinearVelocity=Vector3.zero cl.AssemblyAngularVelocity=Vector3.zero
end

    if not b or not b.Parent then ClearPred() return end

    if sp<8 then ClearPred() return end
    if not predFolder or not predFolder.Parent then
        predFolder=Instance.new("Folder") predFolder.Name="DzPredictionBeam" predFolder.Parent=workspace
    end
    local pts={} local p=b.Position local vv=v
    for i=1,PRED_STEPS do
        vv=vv+PRED_GRAVITY*PRED_DT vv=vv*(1-0.012*PRED_DT*60) p=p+vv*PRED_DT
        if p.Y<=(b.Position.Y-15) and vv.Y<0 then table.insert(pts,Vector3.new(p.X,b.Position.Y-1,p.Z)) break end

    end
    if #pts<2 then ClearPred() return end

    for i=1,nd do

local function ScoreFree(p,bp)
    local r=GetPlayerRoot(p) if not r then return -math.huge end
    local ps=r.Position local db=(ps-bp).Magnitude
    if db<4 or db>120 then return -math.huge end
    local ne=math.huge
    for _,o in ipairs(Players:GetPlayers()) do
        if o~=LP and o~=p and not IsSameTeam(o) then
            local or_=GetPlayerRoot(o) if or_ then local d=(or_.Position-ps).Magnitude if d<ne then ne=d end end
        end
    end
    if ne==math.huge then ne=50 end
    if PassAutoFreeEnabled and ne<PassFreeMinDistance then return -math.huge end
    local ds=1-math.abs(db-25)/50
    local tv=r.AssemblyLinearVelocity local fs=tv.Magnitude>1 and 4 or 0
    return ne*2.2+ds*15+fs
end

local function GuiarPasse(b,dt)
    if not PassEnabled or not b then return end
    local v=b.AssemblyLinearVelocity local mg=Vector3.new(v.X,0,v.Z).Magnitude
    if not PassGuiaAtivo then if PassToolTrg or mg>=VEL_PASSE_MIN then TentarPasse(b) end end
    if not PassGuiaAtivo then return end
    if (not PassNickFilter or PassNickFilter=="") and os.clock()-PassLastRecheck>=PassRecheckInterval then
        PassLastRecheck=os.clock()
        local fr=GetSmartTarget(b.Position)
        if fr and fr~=PassTargetPlayer then PassTargetPlayer=fr Notify("Redirecionado → "..AllyName(fr),"info")
        elseif not fr and PassTargetPlayer then
            local cs=ScoreFree(PassTargetPlayer,b.Position)
            if cs==-math.huge or cs<PassMinTargetScore then
                local rp=GetSmartTarget(b.Position)
                if rp then PassTargetPlayer=rp Notify("Novo alvo: "..AllyName(rp),"info") end
            end
        end
    end
    local r=GetPlayerRoot(PassTargetPlayer) if not r then FinalizarPasse("Alvo sumiu") return end
    local tv=r.AssemblyLinearVelocity local pt=r.Position+tv*PassLeadTime
    local dv=Vector3.new(pt.X-b.Position.X,0,pt.Z-b.Position.Z) local di=dv.Magnitude
    local vh=Vector3.new(v.X,0,v.Z)
    if PassChegou or di<=PASS_RAIO_CHEGADA then
        PassChegou=true
        local nv=b.AssemblyLinearVelocity*freioPosGol
        if nv.Magnitude<3 then
            b.AssemblyLinearVelocity=Vector3.zero b.AssemblyAngularVelocity=Vector3.zero
            PassContandoParada=PassContandoParada+1
            if PassContandoParada>12 then FinalizarPasse("Passe chegou em "..AllyName(PassTargetPlayer)) end return
        end b.AssemblyLinearVelocity=nv return
    end
    local ap=vh.Magnitude>1 and dv.Magnitude>0.1 and vh.Unit:Dot(dv.Unit)>0.35
    if di<=PASS_ZONA_FREIO and ap then
        local nm=math.max(vh.Magnitude-ATRITO_GOAL*dt,0)
        if nm*dt*60<di then
            b.AssemblyLinearVelocity=Vector3.zero b.AssemblyAngularVelocity=Vector3.zero
            PassContandoParada=PassContandoParada+1
            if PassContandoParada>12 then FinalizarPasse("Passe chegou em "..AllyName(PassTargetPlayer)) end return
        end
        b.AssemblyLinearVelocity=Vector3.new(vh.Unit.X*nm,v.Y,vh.Unit.Z*nm) return
    end
    local nm=math.max(mg-ATRITO*dt,VEL_MINIMA*0.85)
    local da=vh.Magnitude>1 and vh.Unit or dv.Unit
    local al=da:Dot(dv.Unit) local de=1-math.max(al,-1)
    local fa=(CORRECAO_ALINHADO+de*CORRECAO_DESALINHADO)*PASS_FORCA_CURVA
    if di<PASS_DIST_INICIO then local t=math.clamp(1-di/PASS_DIST_INICIO,0,1) fa=fa*(0.7+t*0.6) end
    if di<22 then fa=fa*REDUCAO_PERTO end
    local df=da:Lerp(dv.Unit,math.clamp(fa,0,0.45))
    if df.Magnitude>0.01 then df=df.Unit else df=dv.Unit end
    b.AssemblyLinearVelocity=Vector3.new(df.X*nm,v.Y,df.Z*nm) b.AssemblyAngularVelocity=Vector3.zero
end

local function GetAimbotGoal(side,bp)
    local now=os.clock() local c=AimbotGoalCache[side]
    if c and now-c.time<0.35 and c.part and c.part.Parent then return c.part.Position end
    local map=FindMapFolder()
    local ns=side=="Blue" and {"Team1","TeamBlue","Blue","BlueTeam"} or {"Team2","TeamGreen","Green","GreenTeam"}
    local bp_=nil
    for _,n in ipairs(ns) do local f=map:FindFirstChild(n,true) if f then local p=FindGoalIn(f,bp) if p then bp_=p break end end end
    if not bp_ then
        local fb={}
        for _,o in ipairs(map:GetDescendants()) do
            if o:IsA("BasePart") then
                local n=o.Name:lower()
                local isS=side=="Blue" and (n:find("blue",1,true) or n:find("azul",1,true)) or side=="Green" and (n:find("green",1,true) or n:find("verde",1,true))
                if isS and (n:find("goal",1,true) or n:find("gol",1,true) or n:find("net",1,true)) then table.insert(fb,o) end
            end
        end
        local bd=-math.huge for _,o in ipairs(fb) do local d=(o.Position-bp).Magnitude if d>bd then bd=d bp_=o end end

local function TryGKCatch(b)
    if not GoalkeeperAutoCatch or not b or not b.Parent or not Root or not Root.Parent then return end
    if not GoalkeeperRemote or not GoalkeeperRemote.Parent then pcall(function() GoalkeeperRemote=RepStorage:FindFirstChild("CatchBall",true) end) end
    local now=os.clock() if now-GoalkeeperLastCatch<GoalkeeperCatchCooldown then return end
    local bv=b.AssemblyLinearVelocity local ld=math.clamp(GoalkeeperBallLead,0.05,0.45)
    local pp=b.Position+bv*ld
    local cd=(b.Position-Root.Position).Magnitude local pd=(pp-Root.Position).Magnitude
    local di=math.min(cd,pd) if di>GoalkeeperCatchRange then return end
    local s=false
    if GoalkeeperRemote and GoalkeeperRemote.Parent then
        s=pcall(function()
            if GoalkeeperRemote:IsA("RemoteEvent") then GoalkeeperRemote:FireServer(b)
            elseif GoalkeeperRemote:IsA("RemoteFunction") then GoalkeeperRemote:InvokeServer(b) end
        end)
    end
    if not s then pcall(function() TocarBola(b) end) end
    GoalkeeperLastCatch=now
end

        local dl=fl-Root.Position
        if dl.Magnitude>GoalkeeperLineOffset then Hum:Move(dl.Unit,false) end

    TryGKCatch(b)

local function DoFollow(b)

    if not Root or not Root.Parent or not Hum or not Hum.Parent then return end
    local bp=b.Position local bv=b.AssemblyLinearVelocity

    local tp=bp+hv*ld
    local dl=tp-Root.Position local md=Vector3.new(dl.X,0,dl.Z) local mg=md.Magnitude
    if mg>FollowStopDistance and mg<180 then
        local ds=md.Unit

    else followVel=Vector3.zero Hum:Move(Vector3.zero,false) end

local function MakeDrag(g)
    local dr,di,ds,sp
    g.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dr=true ds=i.Position sp=g.Position
            i.Changed:Connect(function() if i.UserInputState==Enum.UserInputState.End then dr=false end end)
        end
    end)
    g.InputChanged:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch then di=i end
    end)
    UIS.InputChanged:Connect(function(i)
        if i==di and dr then
            local d=i.Position-ds g.Position=UDim2.new(sp.X.Scale,sp.X.Offset+d.X,sp.Y.Scale,sp.Y.Offset+d.Y)
        end
    end)
end
