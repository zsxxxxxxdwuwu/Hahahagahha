if type(task) ~= "table" then
    task = {}
end
if type(task.wait) ~= "function" then
    task.wait = function(n)
        n = tonumber(n) or 0
        local rs = game:GetService("RunService")
        if n <= 0 then
            rs.Heartbeat:Wait()
            return
        end
        local t0 = tick()
        while tick() - t0 < n do
            rs.Heartbeat:Wait()
        end
    end
end
if type(task.delay) ~= "function" then
    task.delay = function(n, fn)
        task.spawn(function()
            task.wait(n)
            if type(fn) == "function" then fn() end
        end)
    end
end
if type(task.spawn) ~= "function" then
    task.spawn = function(fn)
        if type(fn) == "function" then
            coroutine.resume(coroutine.create(fn))
        end
    end
end
if type(task.defer) ~= "function" then
    task.defer = function(fn)
        task.spawn(fn)
    end
end

local function SalvarCoordenadas(tabela)
    local c = ""
    for nome, coords in pairs(tabela) do
        c = c .. string.format("%s|%.2f,%.2f,%.2f\n", nome, coords.x, coords.y, coords.z)
    end
    pcall(function() writefile("ball_pro_coords.txt", c) end)
end

local function CarregarCoordenadas()
    local ok, conteudo = pcall(function() return readfile("ball_pro_coords.txt") end)
    if not ok or not conteudo then return {} end
    local t = {}
    for linha in conteudo:gmatch("[^\n]+") do
        local nome, xy = linha:match("^(.+)|(.+)$")
        if nome and xy then
            local x, y, z = xy:match("([^,]+),([^,]+),([^,]+)")
            if x and y and z then
                t[nome] = {x = tonumber(x), y = tonumber(y), z = tonumber(z)}
            end
        end
    end
    return t
end

    tps = true, mps = true, esa = true, mrs = true, prs = true,
    rbz = true, saml = true, ball = true, ballpart = true, csf = true,

local Kers0neBallCache = { Set = {}, LastFullScan = 0, RescanInterval = 1.0, Conn = nil }

local function Kers0neIsBall(part, myChar)
    if not part or not part:IsA("BasePart") then return false end
    if part.Anchored then return false end
    if myChar and part:IsDescendantOf(myChar) then return false end
    local model = part:FindFirstAncestorOfClass("Model")
    if model and model:FindFirstChildOfClass("Humanoid") then return false end
    local nameLower = part.Name:lower()
    if not KERS0NE_BALL_NAMES[nameLower] then
        if not (nameLower:find("ball", 1, true) or nameLower:find("bola", 1, true)) then
            return false
        end
    end
    local size = part.Size
    if size.X < 0.4 or size.X > 8 then return false end
    if size.Y < 0.4 or size.Y > 8 then return false end
    if size.Z < 0.4 or size.Z > 8 then return false end
    return true
end

local function Kers0neCacheMark(inst)
    if inst:IsA("BasePart") then
        local n = inst.Name:lower()
        if KERS0NE_BALL_NAMES[n] or n:find("ball", 1, true) or n:find("bola", 1, true) then
            Kers0neBallCache.Set[inst] = true
        end
    end
end

local function Kers0neFullScan()
    local now = tick()
    if now - Kers0neBallCache.LastFullScan < Kers0neBallCache.RescanInterval then return end
    Kers0neBallCache.LastFullScan = now
    for _, d in ipairs(workspace:GetDescendants()) do
        Kers0neCacheMark(d)
    end
end

if not Kers0neBallCache.Conn then
    Kers0neBallCache.Conn = workspace.DescendantAdded:Connect(Kers0neCacheMark)
end

local function Kers0neCacheClean()
    for k in pairs(Kers0neBallCache.Set) do
        if not k or not k.Parent then
            Kers0neBallCache.Set[k] = nil
        end
    end
end

local function Kers0nePickBall(myRoot, myChar, mode)
    if not myRoot or not myRoot.Parent then return nil end
    Kers0neCacheClean()
    if next(Kers0neBallCache.Set) == nil then Kers0neFullScan() end
    mode = mode or "closest"
    local best, bestScore = nil, -math.huge
    for part in pairs(Kers0neBallCache.Set) do
        if part and part.Parent and Kers0neIsBall(part, myChar) then
            local delta = part.Position - myRoot.Position
            local dist = delta.Magnitude
            local score
            if mode == "fastest" then
                score = part.AssemblyLinearVelocity.Magnitude
            elseif mode == "incoming" then
                local vel = part.AssemblyLinearVelocity
                if vel.Magnitude < 1 then score = -math.huge
                else score = vel.Unit:Dot(-delta.Unit) * 100 - dist * 0.1 end
            else
                score = -dist
            end
            if score > bestScore then bestScore = score; best = part end
        end
    end
    return best
end

local Kers0neFollow = {
    Enabled = false, Mode = "closest",
    StopDistance = 2.5, MaxDistance = 200, MinSpeedToTrack = 0,
}

function Kers0neFollow.Step()
    if not Kers0neFollow.Enabled then return end
    if not RootPart or not RootPart.Parent then return end
    if not Humanoid or not Humanoid.Parent or Humanoid.Health <= 0 then return end
    if IsManualInputActive and IsManualInputActive() then return end
    if Humanoid.Sit or Humanoid.SeatPart then return end
    local state = Humanoid:GetState()
    if state ~= Enum.HumanoidStateType.Running
        and state ~= Enum.HumanoidStateType.RunningNoPhysics
        and state ~= Enum.HumanoidStateType.Jumping
        and state ~= Enum.HumanoidStateType.Landed
        and state ~= Enum.HumanoidStateType.Freefall then
        return
    end
    local ball = Kers0nePickBall(RootPart, Character, Kers0neFollow.Mode)
    if not ball then return end
    local vel = ball.AssemblyLinearVelocity
    if Kers0neFollow.MinSpeedToTrack > 0 and vel.Magnitude < Kers0neFollow.MinSpeedToTrack then return end
    local delta = ball.Position - RootPart.Position
    local flat = Vector3.new(delta.X, 0, delta.Z)
    local dist = flat.Magnitude
    if dist > Kers0neFollow.MaxDistance then return end
    if dist <= Kers0neFollow.StopDistance then
        Humanoid:Move(Vector3.zero, false)
        return
    end
    Humanoid:Move(flat.Unit, false)
end

local cooldownToque = 0

local function CalcularDesvio(bola, dirAtual)
    if not desvioOn then return dirAtual end
    local melhorPlayer, melhorDist = nil, math.huge
    local bolaPos = bola.Position
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and pl.Character then
            local hrp = pl.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local d = (hrp.Position - bolaPos).Magnitude
                if d <= raioDesvio and d < melhorDist then
                    local proPlayer = Vector3.new(hrp.Position.X - bolaPos.X, 0, hrp.Position.Z - bolaPos.Z)
                    if proPlayer.Magnitude > 0.1 then
                        if dirAtual:Dot(proPlayer.Unit) > 0.3 then
                            melhorPlayer = hrp
                            melhorDist = d
                        end
                    end
                end
            end
        end
    end
    if not melhorPlayer then return dirAtual end
    local proPlayer = Vector3.new(melhorPlayer.Position.X - bolaPos.X, 0, melhorPlayer.Position.Z - bolaPos.Z).Unit
    local perp = Vector3.new(-dirAtual.Z, 0, dirAtual.X)
    local lado = perp:Dot(proPlayer) > 0 and -perp or perp
    local urgencia = math.clamp(1 - (melhorDist / raioDesvio), 0, 1)
    local dirDesvio = dirAtual:Lerp(lado, forcaDesvio * urgencia)
    return dirDesvio.Magnitude > 0.01 and dirDesvio.Unit or dirAtual
end

local function GuiarBola(bola, posGoal, dt)
    if not estadoSistema.ativo or not estadoSistema.alvoAtual then return end
    local vel = bola.AssemblyLinearVelocity
    local mag = Vector3.new(vel.X, 0, vel.Z).Magnitude
    if not estadoSistema.guiaAtivo and not estadoSistema.cruzou and mag >= VEL_CHUTE_MIN then
        estadoSistema.guiaAtivo = true
        estadoSistema.contandoParada = 0
        Notify(string.format("Chute detectado! vel: %.0f", mag), "info")
    end
    if not estadoSistema.guiaAtivo then return end
    local distGoal = Vector3.new(posGoal.X - bola.Position.X, 0, posGoal.Z - bola.Position.Z)
    local dist = distGoal.Magnitude
    if estadoSistema.cruzou or dist <= raioGoal then
        estadoSistema.cruzou = true
        local novoVel = bola.AssemblyLinearVelocity * freioPosGol
        if novoVel.Magnitude < 3 then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            estadoSistema.contandoParada = estadoSistema.contandoParada + 1
            if estadoSistema.contandoParada > TEMPO_PARA_DESLIGAR then
                estadoSistema.guiaAtivo = false
                estadoSistema.cruzou = false
                estadoSistema.contandoParada = 0
                estadoSistema.ativo = false
                Notify("Alvo alcançado — sistema desligado", "ok")
            end
            return
        end
        bola.AssemblyLinearVelocity = novoVel
        return
    end
    local velHoriz = Vector3.new(vel.X, 0, vel.Z)
    local apontandoProGoal = velHoriz.Magnitude > 1 and velHoriz.Unit:Dot(distGoal.Unit) > 0.4
    if dist <= zonaGoal and apontandoProGoal then
        local novaMagZ = math.max(velHoriz.Magnitude - ATRITO_GOAL * dt, 0)
        if novaMagZ * dt * 60 < dist then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            estadoSistema.contandoParada = estadoSistema.contandoParada + 1
            if estadoSistema.contandoParada > TEMPO_PARA_DESLIGAR then
                estadoSistema.guiaAtivo = false
                estadoSistema.cruzou = false
                estadoSistema.contandoParada = 0
                estadoSistema.ativo = false
            end
            return
        end
        bola.AssemblyLinearVelocity = Vector3.new(velHoriz.Unit.X * novaMagZ, vel.Y, velHoriz.Unit.Z * novaMagZ)
        return
    end
    local vaiPassar = dist < zonaGoal and velHoriz.Magnitude > 1 and velHoriz.Unit:Dot(distGoal.Unit) > 0.6
    local novaMag = math.max(mag - ATRITO * dt, VEL_MINIMA)
    local dirAtual = velHoriz
    if dirAtual.Magnitude < 1 then dirAtual = distGoal.Unit else dirAtual = dirAtual.Unit end
    local dirFinal = dirAtual
    if not vaiPassar then
        local alinhamento = dirAtual:Dot(distGoal.Unit)
        local desalinhado = 1 - math.max(alinhamento, -1)
        local fator = (CORRECAO_ALINHADO + desalinhado * CORRECAO_DESALINHADO) * FORCA_CURVA
        if dist < DIST_INICIO_CURVA then
            local t = math.clamp(1 - (dist / DIST_INICIO_CURVA), 0, 1)
            fator = fator * (0.7 + t * 0.6)
        end
        if dist < 25 then fator = fator * REDUCAO_PERTO end
        dirFinal = dirAtual:Lerp(distGoal.Unit, math.clamp(fator, 0, 0.45))
        if dirFinal.Magnitude > 0.01 then dirFinal = dirFinal.Unit else dirFinal = distGoal.Unit end
    end
    dirFinal = CalcularDesvio(bola, dirFinal)
    local velY = bola.AssemblyLinearVelocity.Y
    bola.AssemblyLinearVelocity = Vector3.new(dirFinal.X * novaMag, velY, dirFinal.Z * novaMag)
    bola.AssemblyAngularVelocity = Vector3.zero
    if os.clock() - cooldownToque > 0.15 then
        cooldownToque = os.clock()
        TocarBola(bola)
    end
end

local function GetValidBallForFollow()
    if not RootPart then return nil end
    local closestBall, shortestDist = nil, math.huge
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local n = obj.Name:lower()
            if n == "tps" or n == "ball" or n:find("ball") or n:find("soccer") then
                local s = obj.Size
                if s.X >= 0.8 and s.X <= 6.0 then
                    local dist = (obj.Position - RootPart.Position).Magnitude
                    if dist < shortestDist then shortestDist = dist; closestBall = obj end
                end
            end
        end
    end
    return closestBall
end

local function DoAutoFollow(ball)
    if not AutoFollowEnabled or IsManualInputActive() or not ball then return end
    if not RootPart or not RootPart.Parent or not Humanoid or not Humanoid.Parent then return end
    local targetPos = ball.Position
    local delta = targetPos - RootPart.Position
    local moveDirection = Vector3.new(delta.X, 0, delta.Z)
    local mag = moveDirection.Magnitude
    if mag > FollowStopDistance and mag < 150 then
        Humanoid:Move(moveDirection.Unit, false)
        RootPart.CFrame = CFrame.new(RootPart.Position, Vector3.new(targetPos.X, RootPart.Position.Y, targetPos.Z))
    else
        Humanoid:Move(Vector3.zero, false)
    end
end

local function DoSkillMode(dt)
    if not SkillModeEnabled then return end
    if not RootPart or not RootPart.Parent then return end
    local closest = GetValidBallForFollow()
    if not closest or not closest.Parent then return end
    local dist = (closest.Position - RootPart.Position).Magnitude
    if dist > 12 then return end
    skillTimer = skillTimer + (dt or 0.016) * SkillSpeed
    if skillTimer >= 1 then skillTimer = 0; skillPhase = (skillPhase + 1) % 4 end
    local t = skillTimer
    local offset
    if skillPhase == 0 then offset = RootPart.CFrame.LookVector * math.sin(t * math.pi) * 2
    elseif skillPhase == 1 then offset = RootPart.CFrame.RightVector * math.sin(t * math.pi) * 2
    elseif skillPhase == 2 then offset = -RootPart.CFrame.RightVector * math.sin(t * math.pi) * 2
    else offset = -RootPart.CFrame.LookVector * math.sin(t * math.pi) * 2 end
    local targetPos = RootPart.Position + offset + Vector3.new(0, -1.5, 0)
    closest.CFrame = CFrame.new(closest.Position:Lerp(targetPos, 0.3), closest.Position)
    closest.AssemblyLinearVelocity = Vector3.zero
    closest.AssemblyAngularVelocity = Vector3.zero
end

local function UpdatePrediction(bola)
    if not PredictionEnabled or StealthMode then ClearPrediction(); return end
    if not bola or not bola.Parent then ClearPrediction(); return end
    local vel = bola.AssemblyLinearVelocity
    if vel.Magnitude < 8 then ClearPrediction(); return end
    if not predFolder or not predFolder.Parent then
        predFolder = Instance.new("Folder")
        predFolder.Name = "DzPredictionBeam"
        predFolder.Parent = workspace
    end
    local points = {}
    local pos = bola.Position
    local v = vel
    for i = 1, PRED_STEPS do
        v = v + PRED_GRAVITY * PRED_DT
        v = v * (1 - 0.012 * PRED_DT * 60)
        pos = pos + v * PRED_DT
        if pos.Y <= (bola.Position.Y - 15) and v.Y < 0 then
            table.insert(points, Vector3.new(pos.X, bola.Position.Y - 1, pos.Z))
            break
        end
        table.insert(points, pos)
    end
    if #points < 2 then ClearPrediction(); return end
    local needed = #points - 1
    while #predBeams > needed do
        local extra = table.remove(predBeams)
        pcall(function() extra:Destroy() end)
    end
    for i = 1, needed do
        local p0 = points[i]
        local p1 = points[i + 1]
        local beamData = predBeams[i]
        if not beamData or not beamData.part0 or not beamData.part0.Parent then
            local a0 = Instance.new("Part")
            a0.Name = "PredA0"
            a0.Anchored = true
            a0.CanCollide = false
            a0.CanQuery = false
            a0.CanTouch = false
            a0.Transparency = 1
            a0.Size = Vector3.new(0.2, 0.2, 0.2)
            a0.Parent = predFolder
            local a1 = Instance.new("Part")
            a1.Name = "PredA1"
            a1.Anchored = true
            a1.CanCollide = false
            a1.CanQuery = false
            a1.CanTouch = false
            a1.Transparency = 1
            a1.Size = Vector3.new(0.2, 0.2, 0.2)
            a1.Parent = predFolder
            local att0 = Instance.new("Attachment", a0)
            local att1 = Instance.new("Attachment", a1)
            local beam = Instance.new("Beam")
            beam.Attachment0 = att0
            beam.Attachment1 = att1
            beam.Width0 = 0.35
            beam.Width1 = 0.15
            beam.FaceCamera = true
            beam.LightEmission = 0.6
            beam.LightInfluence = 0
            beam.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 60, 60)),

                NumberSequenceKeypoint.new(1, 0.55)
            })

        end
        beamData.part0.CFrame = CFrame.new(p0)
        beamData.part1.CFrame = CFrame.new(p1)
    end
end

local function FindAllyByPartialNick(partial)
    if not partial or partial == "" then return nil, "vazio" end
    local q = partial:lower()
    local matches = {}
    if TestBotRoot and TestBotRoot.Parent then
        local botName = "robo kkkkkk"
        if botName:sub(1, #q) == q or botName:find(q, 1, true) then
            table.insert(matches, "TEST_BOT")
        end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsSameTeam(plr) then
            local name = plr.Name:lower()
            local display = (plr.DisplayName or ""):lower()
            if name:sub(1, #q) == q or name:find(q, 1, true) or display:sub(1, #q) == q or display:find(q, 1, true) then
                table.insert(matches, plr)
            end
        end
    end
    if #matches == 0 then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local name = plr.Name:lower()
                local display = (plr.DisplayName or ""):lower()
                if name:sub(1, #q) == q or name:find(q, 1, true) or display:sub(1, #q) == q or display:find(q, 1, true) then
                    table.insert(matches, plr)
                end
            end
        end
    end
    if #matches == 1 then return matches[1], "ok" end
    if #matches == 0 then return nil, "nenhum" end
    return nil, "multiplos", matches
end

local function ScoreAllyFreedom(plr, ballPos)
    local root = GetPlayerRoot(plr)
    if not root then return -math.huge end
    local pos = root.Position
    local distBall = (pos - ballPos).Magnitude
    if distBall < 4 or distBall > 120 then return -math.huge end
    local nearestEnemy = math.huge
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= LocalPlayer and other ~= plr and not IsSameTeam(other) then
            local oroot = GetPlayerRoot(other)
            if oroot then
                local d = (oroot.Position - pos).Magnitude
                if d < nearestEnemy then nearestEnemy = d end
            end
        end
    end
    if nearestEnemy == math.huge then nearestEnemy = 50 end
    local distScore = 1 - math.abs(distBall - 25) / 50
    return nearestEnemy * 2 + distScore * 15
end

local function GetBestAlly(ballPos)
    if PassNickFilter and PassNickFilter ~= "" then
        local plr, status = FindAllyByPartialNick(PassNickFilter)
        if plr then return plr end
        if status == "multiplos" then return nil end
    end
    local best, bestScore = nil, -math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and IsSameTeam(plr) then
            local score = ScoreAllyFreedom(plr, ballPos)
            if score > bestScore then bestScore = score; best = plr end
        end
    end
    if not best then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local score = ScoreAllyFreedom(plr, ballPos)
                if score > bestScore then bestScore = score; best = plr end
            end
        end
    end
    return best
end

local function GuiarPasse(bola, dt)
    if not PassEnabled or not bola then return end
    local vel = bola.AssemblyLinearVelocity
    local mag = Vector3.new(vel.X, 0, vel.Z).Magnitude
    if not PassGuiaAtivo then
        if PassToolTriggered or mag >= VEL_PASSE_MIN then TentarIniciarPasse(bola) end
    end
    if not PassGuiaAtivo then return end
    local root = GetPlayerRoot(PassTargetPlayer)
    if not root then FinalizarPasse("Alvo sumiu"); return end
    local posGoal = root.Position
    local distVec = Vector3.new(posGoal.X - bola.Position.X, 0, posGoal.Z - bola.Position.Z)
    local dist = distVec.Magnitude
    local velH = Vector3.new(vel.X, 0, vel.Z)
    if PassChegou or dist <= PASS_RAIO_CHEGADA then
        PassChegou = true
        local novoVel = bola.AssemblyLinearVelocity * freioPosGol
        if novoVel.Magnitude < 3 then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            PassContandoParada = PassContandoParada + 1
            if PassContandoParada > 12 then FinalizarPasse("Passe chegou em " .. AllyName(PassTargetPlayer)) end
            return
        end
        bola.AssemblyLinearVelocity = novoVel
        return
    end
    local apontando = velH.Magnitude > 1 and distVec.Magnitude > 0.1 and velH.Unit:Dot(distVec.Unit) > 0.35
    if dist <= PASS_ZONA_FREIO and apontando then
        local novaMagZ = math.max(velH.Magnitude - ATRITO_GOAL * dt, 0)
        if novaMagZ * dt * 60 < dist then
            bola.AssemblyLinearVelocity = Vector3.zero
            bola.AssemblyAngularVelocity = Vector3.zero
            PassContandoParada = PassContandoParada + 1
            if PassContandoParada > 12 then FinalizarPasse("Passe chegou em " .. AllyName(PassTargetPlayer)) end
            return
        end
        bola.AssemblyLinearVelocity = Vector3.new(velH.Unit.X * novaMagZ, vel.Y, velH.Unit.Z * novaMagZ)
        return
    end
    local novaMag = math.max(mag - ATRITO * dt, VEL_MINIMA * 0.85)
    local dirAtual = velH.Magnitude > 1 and velH.Unit or distVec.Unit
    local alinhamento = dirAtual:Dot(distVec.Unit)
    local desalinhado = 1 - math.max(alinhamento, -1)
    local fator = (CORRECAO_ALINHADO + desalinhado * CORRECAO_DESALINHADO) * PASS_FORCA_CURVA
    if dist < PASS_DIST_INICIO then
        local t = math.clamp(1 - (dist / PASS_DIST_INICIO), 0, 1)
        fator = fator * (0.7 + t * 0.6)
    end
    if dist < 22 then fator = fator * REDUCAO_PERTO end
    local dirFinal = dirAtual:Lerp(distVec.Unit, math.clamp(fator, 0, 0.45))
    if dirFinal.Magnitude > 0.01 then dirFinal = dirFinal.Unit else dirFinal = distVec.Unit end
    bola.AssemblyLinearVelocity = Vector3.new(dirFinal.X * novaMag, vel.Y, dirFinal.Z * novaMag)
    bola.AssemblyAngularVelocity = Vector3.zero
end

local Kers0neDisguise = {
    Enabled = true, PassDamp = 0.72, ShotDamp = 0.78,
    JitterPass = 0.015, JitterShot = 0.010,
    HumanPulseAmp = 0.006, HumanPulseHz = 3.7,
}

local function Kers0neApplyDisguise(bola, damp, jitter, tag)
    if not bola or not bola.Parent then return end
    local vel = bola.AssemblyLinearVelocity
    local velH = Vector3.new(vel.X, 0, vel.Z)
    local mag = velH.Magnitude
    if mag < 3 then return end
    local dir = velH.Unit
    local t = tick()
    local pulse = math.sin(t * Kers0neDisguise.HumanPulseHz) * Kers0neDisguise.HumanPulseAmp
    local perp = Vector3.new(-dir.Z, 0, dir.X)
    local humanized = dir + perp * pulse
    if humanized.Magnitude < 0.01 then humanized = dir else humanized = humanized.Unit end
    if jitter > 0 then
        humanized = Vector3.new(
            humanized.X + (math.random() - 0.5) * jitter,
            humanized.Y,
            humanized.Z + (math.random() - 0.5) * jitter
        )
        if humanized.Magnitude > 0.01 then humanized = humanized.Unit end
    end
    local softDir = dir:Lerp(humanized, damp)
    if softDir.Magnitude < 0.01 then softDir = dir else softDir = softDir.Unit end
    bola.AssemblyLinearVelocity = Vector3.new(softDir.X * mag, vel.Y, softDir.Z * mag)
end

local function MakeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function RefreshPassTargetLabel()
    PassNickFilter = (passNickBox.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if PassNickFilter == "" then
        lblPassTarget.Text = "Alvo: automático (mais livre)"
        return
    end
    local plr, status, matches = FindAllyByPartialNick(PassNickFilter)
    if plr then
        lblPassTarget.Text = "Alvo: " .. AllyName(plr)
        lblPassTarget.TextColor3 = Color3.fromRGB(80, 255, 120)
    elseif status == "multiplos" then
        local names = {}
        for i, m in ipairs(matches or {}) do
            if i <= 3 then table.insert(names, AllyName(m)) end
        end
        lblPassTarget.Text = "Vários: " .. table.concat(names, ", ")
        lblPassTarget.TextColor3 = Color3.fromRGB(255, 180, 40)
    else
        lblPassTarget.Text = "Nenhum jogador com \"" .. PassNickFilter .. "\""
        lblPassTarget.TextColor3 = Color3.fromRGB(255, 80, 80)
    end
end

local function TogglePass()
    PassNickFilter = (passNickBox.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if not PassEnabled then
        if PassNickFilter ~= "" then
            local plr, status = FindAllyByPartialNick(PassNickFilter)
            if status == "nenhum" then Notify("Nick não encontrado", "warn"); return end
            if status == "multiplos" then Notify("Vários nicks", "warn"); return end
            PassTargetPlayer = plr
            Notify("Passe travado em " .. AllyName(plr), "ok")
        else
            PassTargetPlayer = nil
            Notify("Passe no aliado mais livre", "ok")
        end
        PassEnabled = true
        PassGuiaAtivo = false
        PassToolTriggered = false
    else
        PassEnabled = false
        PassGuiaAtivo = false
        PassChegou = false
        PassContandoParada = 0
        PassTargetPlayer = nil
        PassToolTriggered = false
        Notify("Passe Inteligente OFF", "info")
    end
    AtualizarVisualPasse()
end
